--[[ai slop documentation because I hate making documentation
================================================================================
    SQL Library (Client-Routed SQLite Wrapper)
    Author: ByakuyaKuchiki
    SQLite version: 3.26.0 (https://sqlite.org/docs.html)
================================================================================

    A secure wrapper allowing both Server and Client realms to interact with 
    GMod's native SQLite client database via 'easynet'. 

    > CRITICAL SECURITY NOTE:
    > This library operates EXCLUSIVELY on the Server and the Chip Owner's Client.
    > If a non-owner client triggers or attempts to execute this script, it 
    > will abort automatically to prevent database tampering.

    ----------------------------------------------------------------------------
    QUICK START EXAMPLES
    ----------------------------------------------------------------------------
    
    [Parameterized Query with Callback]
    local queryString = "SELECT level, tokens FROM user_stats WHERE steamid = ? AND active = ?;"
    local parameters  = { "STEAM_0:1:123456", true }

    SQL:query(queryString, parameters, function(results)
        if not results then 
            print("No data found or query failed.") 
            return 
        end
        
        -- Rows are returned as an array of dictionaries:
        for i, row in ipairs(results) do
            print(string.format("Level: %s | Tokens: %s", row.level, row.tokens))
        end
    end)

    ----------------------------------------------------------------------------
    API REFERENCE
    ----------------------------------------------------------------------------

    [Initialization & State]

    * SQL:isInitialized() -> [Boolean]
        Returns true if the database channel is awake and ready. 
        - Client realm defaults immediately to `true`.
        - Server realm waits dynamically until the owner client finishes loading.

    * SQL:getHistory() -> [Table]
        Returns an ordered array of all raw query strings processed during this 
        session, with the most recent query at index `1`.

    [Database Operations]

    * SQL:query(queryStr, valuesTable, callback)
        Prepares and executes an SQLite query string.
        - queryStr    [String]  : The SQL statement. Use '?' for safe parameters.
        - valuesTable [Table]   : Array of values to sequentially replace every '?'.
                                  Pass an empty table `{}` if no values are needed.
        - callback    [Function]: Optional function fired when data returns.
                                  Receives a [Table] of rows or `nil` on failure.

    * SQL:exists(tableName, callback)
        Checks if a specific table exists in the database.
        - tableName   [String]  : The name of the table to verify.
        - callback    [Function]: Receives a [Boolean] indicating existence.

    * SQL:remove(tableName, callback)
        Drops/removes a table from the client-side database safely.
        - tableName   [String]  : The name of the table to drop.
        - callback    [Function]: Receives a [Boolean] indicating completion status.

    * SQL:safeString(text, noQuotes, callback)
        Escapes a string securely for raw SQL insertion.
        - text        [String]  : The text string to escape.
        - noQuotes    [Boolean] : Optional. If true, omits enclosing quotes. Defaults to true.
        - callback    [Function]: Receives the securely escaped [String].

    ----------------------------------------------------------------------------
    AUTOMATIC TYPE ESCAPING RULES FOR '?'
    ----------------------------------------------------------------------------
    * Strings  -> Automatically sanitized and wrapped in single quotes (e.g. 'Byakuya')
    * Booleans -> Converted into standard SQL integer bit-flags ('1' or '0')
    * Others   -> (Numbers, Vectors, Angles, etc.) Cast cleanly via tostring()

==============================================================================]]
--@name SQL Lib
--@author ByakuyaKuchiki
--@shared

local old_net
local net = net
if not easynet then
    old_net = table.copy(net)
    --@include byalib/shared/easynet.txt
    net = require("byalib/shared/easynet.txt")
end

local me = owner()
if CLIENT and player() ~= me then return end 

SQL = {}
SQL.initialized = false
SQL.history = {}
local callbacks = {}

if CLIENT then SQL.initialized = true
elseif SERVER then hook.add("ClientInitialized", "SQL:WAITINGFOROWNER", function(ply) if ply == me then SQL.initialized = true end end); SQL.initQueue = {} end
    
function SQL:isInitialized() return SQL.initialized end
function SQL:getHistory() return SQL.history end
local function addHistory(query) table.insert(SQL.history, 1, query) end

function SQL:exists(name, callback)
    if CLIENT then callback(sql.tableExists(name))
    elseif SERVER then 
        local hash = crc(name)
        if not SQL.initialized then --wait for client
            table.insert(SQL.initQueue, {"exists", name, hash})
        else --client is initialized
            net.send(me, "SQL:EXISTS", name, hash)
        end
        
        if not callback then return end 
        callbacks[hash] = callback
    end
end

function SQL:remove(name, callback)
    if CLIENT then callback(sql.tableRemove(name))
    elseif SERVER then 
        local hash = crc(name)
        if not SQL.initialized then --wait for client
            table.insert(SQL.initQueue, {"remove", name, hash})
        else --client is initialized
            net.send(me, "SQL:EXISTS", name, hash)
        end
        
        if not callback then return end 
        callbacks[hash] = callback
    end
end

--https://github.com/Facepunch/garrysmod/blob/master/garrysmod/lua/includes/util/sql.lua#L27-L27
function SQL:safeString(text, noQuotes, callback)
    text = text:gsub("'","''")
    
    local null_chr = string.find(text, "\0")
    if null_chr then
        text = string.sub(text, 1, null_chr - 1)
    end
    
    if noQuotes then
        return text
    end
    
    local newStr = "'" .. text .. "'"
    if callback then callback(newStr) end
    return newStr
end

local function formatQuery(queryString, params)
    local index = 0
    
    local formatted = string.gsub(queryString, "%?", function()
        index = index + 1
        local val = params[index]
        
        if val == nil then return error("missing value for placeholder #" .. index) end 
        
        if isstring(val) then
            return string.format("'%s'", SQL:safeString(val, true), true)
        elseif isbool(val) then
            return val and "1" or "0" 
        else
            return tostring(val) 
        end
    end)
    
    return formatted
end

function SQL:query(query, values, callback)
    if values then query = formatQuery(query, values) end
    
    if CLIENT then
        local results = sql.query(query)
        if callback then callback(results) end
        addHistory(query)
    elseif SERVER then
        local hash = crc(query)
        
        if not SQL.initialized then --wait for client
            table.insert(SQL.initQueue, {"query", query, hash})
        else --client is initialized
            net.send(me, "SQL:QUERY", query, hash)
        end
        
        addHistory(query)
        if not callback then return end 
        callbacks[hash] = callback
    end
end

--handle queueing
if SERVER then
    hook.add("Think", "initQueue", coroutine.wrap(function()
        while not SQL.initialized do coroutine.yield() end
        if table.isEmpty(SQL.initQueue) then hook.remove("Think", "initQueue"); return end
        
        for _, tbl in ipairs(SQL.initQueue) do
            local type,value,hash = unpack(tbl)
            
            if type == "query" then
                net.send(me, "SQL:QUERY", value, hash)
            elseif type == "exists" then
                net.send(me, "SQL:EXISTS", value, hash)
            elseif type == "remove" then
                net.send(me, "SQL:REMOVE", value, hash)
            end
        end
        
        hook.remove("Think", "initQueue")
    end))
end

if CLIENT then
    net.addCallback("SQL:QUERY", function(query, hash)
        addHistory(query)
        local results = sql.query(query)
        net.send("SQL:QUERYRESULTS", hash, results)
    end)
    net.addCallback("SQL:EXISTS", function(name, hash)
        local exists = sql.tableExists(name)
        net.send("SQL:EXISTSRESULTS", exists, hash)
    end)
    net.addCallback("SQL:REMOVE", function(name, hash)
        local wasRemoved = sql.tableRemove(name)
        net.send("SQL:REMOVERESULTS", wasRemoved, hash)
    end)
elseif SERVER then
    --all the callbacks are the same for the most part, run in a for loop
    for _, type in ipairs({"QUERY", "EXISTS", "REMOVE"}) do
        net.addCallback(("SQL:%sRESULTS"):format(type), function(_, ...)
            local value, hash = ...
            if type == "QUERY" then hash,value = value,hash end --query swaps these values
            if not callbacks[hash] then return end 
            callbacks[hash](value)
            callbacks[hash] = nil
        end)
    end
end

if old_net then net = old_net end
